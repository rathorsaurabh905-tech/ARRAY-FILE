#include <stdio.h>

int main() {
    int a[2][3] = {{5, 6, 7}, {8, 9, 10}};
    int b[2][3] = {{5, 6, 7}, {8, 9, 10}};

    printf("Addition of array:\n");

    for (int i = 0; i < 2; i++) {
        for (int j = 0; j < 3; j++) {
            printf("%d ", a[i][j] + b[i][j]);
        }
        printf("\n");
    }

    printf("\n");

    return 0;
}
